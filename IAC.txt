# -*- mode: ruby -*-
# vi: set ft=ruby :

Vagrant.configure("2") do |config|
  DATA     = Time.now.strftime("%Y%m%d")
  VM_NAME  = "oracle-6-#{DATA}"
  VM_IP    = "192.168.56.36"
  VM_USER  = "tenable"
  VM_PASS  = "C9p5au8naa@2026"

  config.vm.box = "generic/oracle6"
  config.vm.hostname = VM_NAME
  config.vm.network "private_network", ip: VM_IP

  config.vm.provider "virtualbox" do |vb|
    vb.name   = VM_NAME
    vb.memory = 2048
    vb.cpus   = 1
    vb.customize ["modifyvm", :id, "--vram", "64"]
    vb.customize ["modifyvm", :id, "--graphicscontroller", "vmsvga"]
  end

  config.ssh.insert_key = true
  config.ssh.forward_agent = false

  config.vm.provision "shell", privileged: true, inline: <<-SHELL
    set -e

    if [ -f /usr/share/zoneinfo/America/Sao_Paulo ]; then
      cp -f /usr/share/zoneinfo/America/Sao_Paulo /etc/localtime
      printf 'ZONE="America/Sao_Paulo"\nUTC=true\nARC=false\n' > /etc/sysconfig/clock
    fi

    if ! getent group wheel >/dev/null 2>&1; then
      groupadd wheel
    fi

    if ! id -u "#{VM_USER}" >/dev/null 2>&1; then
      useradd -m -s /bin/bash "#{VM_USER}"
    fi

    echo "#{VM_USER}:#{VM_PASS}" | chpasswd
    usermod -aG wheel "#{VM_USER}"

    if command -v sudo >/dev/null 2>&1; then
      mkdir -p /etc/sudoers.d
      grep -Eq '^[[:space:]]*#includedir[[:space:]]+/etc/sudoers.d' /etc/sudoers || \
        echo '#includedir /etc/sudoers.d' >> /etc/sudoers
      cat > /etc/sudoers.d/90-tenable-scan << EOF
Defaults:#{VM_USER} !requiretty
#{VM_USER} ALL=(ALL) NOPASSWD: ALL
EOF
      chmod 440 /etc/sudoers.d/90-tenable-scan
    fi

    if [ -f /etc/ssh/sshd_config ]; then
      sed -ri 's/^[#[:space:]]*PasswordAuthentication[[:space:]].*/PasswordAuthentication yes/' /etc/ssh/sshd_config
      sed -ri 's/^[#[:space:]]*PubkeyAuthentication[[:space:]].*/PubkeyAuthentication yes/' /etc/ssh/sshd_config
      grep -Eq '^[[:space:]]*PasswordAuthentication[[:space:]]+yes' /etc/ssh/sshd_config || echo 'PasswordAuthentication yes' >> /etc/ssh/sshd_config
      grep -Eq '^[[:space:]]*PubkeyAuthentication[[:space:]]+yes' /etc/ssh/sshd_config || echo 'PubkeyAuthentication yes' >> /etc/ssh/sshd_config
      service sshd reload >/dev/null 2>&1 || service sshd restart >/dev/null 2>&1 || true
    fi

    HOME_DIR="/home/#{VM_USER}"
    SSH_DIR="$HOME_DIR/.ssh"
    AUTH_KEYS="$SSH_DIR/authorized_keys"

    mkdir -p "$SSH_DIR"
    touch "$AUTH_KEYS"

    if grep -qxF "ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIBQosqCek+0A91VxyYJNES7bRRo3FxaXcxsD4gEJRgEo renatoroberto25" "$AUTH_KEYS"; then
      true
    else
      echo "ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIBQosqCek+0A91VxyYJNES7bRRo3FxaXcxsD4gEJRgEo renatoroberto25" >> "$AUTH_KEYS"
    fi

    chown -R #{VM_USER}:#{VM_USER} "$SSH_DIR"
    chmod 700 "$SSH_DIR"
    chmod 600 "$AUTH_KEYS"

    cat > /etc/issue.net << 'EOF'
************************************************************
* ACESSO RESTRITO — SISTEMA MONITORADO HITSS        *
************************************************************
Uso exclusivo de pessoal autorizado.
EOF
    chmod 644 /etc/issue.net

    echo "[OK] Estrutura básica pronta."
  SHELL
end
